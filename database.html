ฐานข้อมูล

ใช้ Google Sheets ของคุณเป็นที่เก็บข้อมูล ความคืบหน้าและร่างสัญญาจะตามคุณไปทุกเครื่อง ตั้งค่าครั้งเดียวใช้เวลาไม่กี่นาที

Google Sheets + Apps Scriptต้องมี PIN

สถานะการเชื่อมต่อ

–

ตั้งค่าขั้นสูง: URL ของสคริปต์

ตั้งค่าครั้งเดียว

URL ที่ให้มาตอนนี้ยังไม่มีฟังก์ชัน doGet (ระบบตอบ "ไม่พบฟังก์ชันของสคริปต์") จึงยังเก็บข้อมูลไม่ได้ ต้องวางโค้ดด้านล่างแล้วทำให้ใช้งานเวอร์ชันใหม่ก่อน
  1. วางโค้ด เปิดโปรเจ็กต์ Apps Script ของ URL นี้ ลบโค้ดใน Code.gs ทั้งหมด วางโค้ดจากกล่องด้านล่าง แล้วกดบันทึก
  2. รัน setup ครั้งเดียว เลือกฟังก์ชัน setup กด "เรียกใช้" อนุญาตสิทธิ์เมื่อระบบถาม จากนั้นดู PIN ใน "บันทึกการเรียกใช้งาน" (หรือตั้งเองที่ ตั้งค่าโปรเจ็กต์ → คุณสมบัติของสคริปต์ → PIN) ระบบจะสร้างชีตชื่อ Math-Online DB ในไดรฟ์ของคุณ
  3. ทำให้ใช้งานเวอร์ชันใหม่ ทำให้ใช้งานได้ → จัดการการทำให้ใช้งานได้ → ไอคอนปากกา → เวอร์ชัน: "เวอร์ชันใหม่" → ทำให้ใช้งานได้ ตั้งค่า เรียกใช้ในฐานะ: ฉัน และ ผู้มีสิทธิ์เข้าถึง: ทุกคน (URL เดิมยังใช้ได้)
  4. เชื่อมต่อ กลับมาที่หน้านี้ ใส่ PIN แล้วกด "เชื่อมต่อ" ถ้าขึ้น "เชื่อมต่อแล้ว" ใช้ได้เลย
/**
 * Math-Online DB
 * ใช้ Google Sheets เป็นฐานข้อมูลเอกสาร JSON ขนาดเล็ก
 * - 1 แท็บ (sheet) = 1 collection  คอลัมน์: id | json | updatedAt
 * - ทุกคำสั่งต้องส่ง PIN (เก็บใน Script Properties ชื่อ PIN)
 * - เรียกผ่าน POST (JSON เป็น text/plain) เท่านั้น ส่วน GET ใช้ตรวจว่าบริการทำงานอยู่
 *
 * ขั้นตอนติดตั้ง
 * 1) วางโค้ดนี้แทนที่ Code.gs ทั้งหมด แล้วกดบันทึก
 * 2) เลือกฟังก์ชัน setup แล้วกด "เรียกใช้" ครั้งเดียว (อนุญาตสิทธิ์เมื่อระบบถาม)
 *    ดูรหัส PIN ใน "บันทึกการเรียกใช้งาน" (Execution log) หรือตั้งเองที่
 *    ตั้งค่าโปรเจ็กต์ > คุณสมบัติของสคริปต์ > PIN
 * 3) ทำให้ใช้งานได้: ทำให้ใช้งานได้ > จัดการการทำให้ใช้งานได้ > ปากกา (แก้ไข)
 *    > เวอร์ชัน: "เวอร์ชันใหม่" > ทำให้ใช้งานได้ (URL เดิมยังใช้ได้)
 *    ประเภท: เว็บแอป · เรียกใช้ในฐานะ: ฉัน · ผู้มีสิทธิ์เข้าถึง: ทุกคน
 */

var MAX_DOC = 45000;       // จำนวนอักขระสูงสุดต่อเอกสาร (เซลล์ Sheets จุได้ ~50,000)
var MAX_LIST = 200;        // จำนวนเอกสารสูงสุดที่คืนในคำสั่ง list
var NAME_RE = /^[A-Za-z0-9_-]{1,40}$/;
var ID_RE = /^[A-Za-z0-9_.-]{1,80}$/;

/** รันครั้งเดียวจากตัวแก้ไข เพื่อสร้างชีตและ PIN */
function setup() {
  var p = PropertiesService.getScriptProperties();
  if (!p.getProperty('PIN')) {
    var pin = Utilities.getUuid().replace(/-/g, '').slice(0, 8);
    p.setProperty('PIN', pin);
    Logger.log('สร้าง PIN ใหม่: ' + pin + '  (เปลี่ยนได้ที่ คุณสมบัติของสคริปต์ > PIN)');
  } else {
    Logger.log('มี PIN อยู่แล้ว (ดูที่ คุณสมบัติของสคริปต์ > PIN)');
  }
  var ss = spreadsheet_();
  Logger.log('ชีตฐานข้อมูล: ' + ss.getUrl());
}

function doGet() {
  return out_({ ok: true, service: 'math-online-db', time: new Date().toISOString() });
}

function doPost(e) {
  try {
    var req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (!auth_(req.pin)) return out_({ ok: false, error: 'PIN ไม่ถูกต้องหรือยังไม่ได้ตั้งค่า' });
    switch (req.action) {
      case 'ping':   return out_({ ok: true });
      case 'get':    return out_(get_(req));
      case 'set':    return out_(set_(req));
      case 'list':   return out_(list_(req));
      case 'delete': return out_(delete_(req));
      default:       return out_({ ok: false, error: 'ไม่รู้จักคำสั่ง: ' + req.action });
    }
  } catch (err) {
    return out_({ ok: false, error: String(err && err.message || err) });
  }
}

/* ---------- คำสั่ง ---------- */
function get_(req) {
  var sh = sheet_(req.collection, false);
  if (!sh) return { ok: true, doc: null };
  var row = findRow_(sh, checkId_(req.id));
  return { ok: true, doc: row ? rowToDoc_(sh, row) : null };
}

function set_(req) {
  var id = checkId_(req.id);
  var json = JSON.stringify(req.data === undefined ? null : req.data);
  if (json.length > MAX_DOC) throw new Error('ข้อมูลใหญ่เกินไป (' + json.length + ' อักขระ)');
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = sheet_(req.collection, true);
    var row = findRow_(sh, id);
    var now = Date.now();
    if (row) {
      var cur = Number(sh.getRange(row, 3).getValue()) || 0;
      if (req.ifVersion != null && Number(req.ifVersion) !== cur) {
        return { ok: false, conflict: true, error: 'มีการแก้ไขใหม่กว่า', doc: rowToDoc_(sh, row) };
      }
      sh.getRange(row, 2, 1, 2).setValues([[json, now]]);
    } else {
      sh.appendRow([id, json, now]);
    }
    return { ok: true, id: id, updatedAt: now };
  } finally {
    lock.releaseLock();
  }
}

function list_(req) {
  var sh = sheet_(req.collection, false);
  if (!sh) return { ok: true, docs: [] };
  var n = sh.getLastRow() - 1;
  if (n < 1) return { ok: true, docs: [] };
  var vals = sh.getRange(2, 1, Math.min(n, MAX_LIST), 3).getValues();
  return { ok: true, docs: vals.map(function (r) { return { id: String(r[0]), data: parse_(r[1]), updatedAt: Number(r[2]) || 0 }; }) };
}

function delete_(req) {
  var id = checkId_(req.id);
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = sheet_(req.collection, false);
    if (!sh) return { ok: true, deleted: false };
    var row = findRow_(sh, id);
    if (row) sh.deleteRow(row);
    return { ok: true, deleted: !!row };
  } finally {
    lock.releaseLock();
  }
}

/* ---------- ตัวช่วย ---------- */
function auth_(pin) {
  var real = PropertiesService.getScriptProperties().getProperty('PIN');
  if (!real || pin == null) return false;
  pin = String(pin);
  if (pin.length !== real.length) return false;
  var diff = 0;
  for (var i = 0; i < pin.length; i++) diff |= pin.charCodeAt(i) ^ real.charCodeAt(i);
  return diff === 0;
}

function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function checkId_(id) {
  if (typeof id !== 'string' || !ID_RE.test(id)) throw new Error('id ต้องเป็นตัวอักษรอังกฤษ ตัวเลข _ - . ไม่เกิน 80 ตัว');
  return id;
}

function spreadsheet_() {
  var p = PropertiesService.getScriptProperties();
  var sid = p.getProperty('SHEET_ID');
  if (sid) return SpreadsheetApp.openById(sid);
  var ss = SpreadsheetApp.create('Math-Online DB');
  p.setProperty('SHEET_ID', ss.getId());
  return ss;
}

function sheet_(name, create) {
  if (typeof name !== 'string' || !NAME_RE.test(name)) throw new Error('ชื่อ collection ต้องเป็นอังกฤษ ตัวเลข _ - ไม่เกิน 40 ตัว');
  var ss = spreadsheet_();
  var sh = ss.getSheetByName(name);
  if (!sh && create) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, 3).setValues([['id', 'json', 'updatedAt']]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function findRow_(sh, id) {
  var n = sh.getLastRow() - 1;
  if (n < 1) return 0;
  var ids = sh.getRange(2, 1, n, 1).getValues();
  for (var i = 0; i < ids.length; i++) if (String(ids[i][0]) === id) return i + 2;
  return 0;
}

function rowToDoc_(sh, row) {
  var v = sh.getRange(row, 1, 1, 3).getValues()[0];
  return { id: String(v[0]), data: parse_(v[1]), updatedAt: Number(v[2]) || 0 };
}

function parse_(s) {
  try { return JSON.parse(s); } catch (e) { return null; }
}

อะไรถูกเก็บในฐานข้อมูล

ข้อมูลอยู่ที่ไหนเมื่อไรที่บันทึก
ความคืบหน้าแผน 30 วันและสิ่งที่มีพร้อมชีต state เอกสาร mo-state-v1ทุกครั้งที่ทำเครื่องหมาย และดึงล่าสุดตอนเปิดหน้าแรก
ติดตามการผลิตคอร์สตะลุยข้อสอบชีต state เอกสาร ex-track-v1ทุกครั้งที่ทำเครื่องหมาย
ร่างสัญญา (ไม่มีเลขประจำตัวประชาชน)ชีต drafts เอกสาร contractเมื่อกด "บันทึกร่างลงฐานข้อมูล" ในหน้าร่างสัญญา

ถ้าเปิดเว็บโดยไม่ได้เชื่อมต่อ ทุกอย่างยังใช้ได้และบันทึกในเบราว์เซอร์ตามเดิม เมื่อเชื่อมแล้วระบบเทียบเวลาและใช้ฉบับที่ใหม่กว่า

เอกสารในฐานข้อมูลตอนนี้

ความปลอดภัย

  • URL ของสคริปต์อยู่ในโค้ดเว็บที่เป็นสาธารณะ ใครก็เรียกได้ แต่ทุกคำสั่งต้องมี PIN ถ้า PIN ผิดจะไม่ได้ข้อมูลใด ๆ
  • PIN เก็บเฉพาะในเบราว์เซอร์ที่คุณกรอก ไม่อยู่ในโค้ดและไม่ถูก push ขึ้น GitHub ใช้ PIN ที่ยาวและไม่ซ้ำกับรหัสอื่น เปลี่ยนได้ทุกเมื่อที่ คุณสมบัติของสคริปต์ → PIN
  • อย่าเก็บเลขประจำตัวประชาชน ชื่อหรือคะแนนของเด็ก ในฐานข้อมูลนี้ ร่างสัญญาเก็บแค่ชื่อและที่อยู่ที่คุณกรอกเองในช่องที่ให้
  • ชีต Math-Online DB อยู่ในบัญชี Google ของคุณ แชร์ชีตเฉพาะคนที่ต้องดู
